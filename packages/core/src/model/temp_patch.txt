      'UPDATE providers SET is_current_global = 0 WHERE is_global = 1'
    ));
    this.statements.set('provider_set_current_global', this.db.prepare(
      "UPDATE providers SET is_current_global = 1, updated_at = CURRENT_TIMESTAMP WHERE id = ? AND COALESCE(tool_id, 'global') = COALESCE(?, 'global')"
    ));

    this.statements.set('provider_unset_current_tool', this.db.prepare(

    ));
    this.statements.set('provider_set_current_tool', this.db.prepare(
      'UPDATE providers SET is_current_tool = 1, updated_at = CURRENT_TIMESTAMP WHERE id = ? AND tool_id = ?'
